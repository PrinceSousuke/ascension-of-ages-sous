;(function () {
  const BASE_STAGE = 'age0'
  const IRON_STAGE = 'age1'
  const TEST_STAGE = 'age2_testing'
  const LEGACY_STAGES = ['age0_primitive', 'age1_unlocked', 'age1_ironera']
  const IRON_EQUIVALENT_STAGES = [
    IRON_STAGE,
    'age1_unlocked',
    'age1_ironera'
  ]
  const HIGHER_EQUIVALENT_STAGES = [
    TEST_STAGE,
    'age2',
    'age3',
    'age4',
    'age5',
    'age6',
    'age7'
  ]

  function runAStagesCommand(player, subcommand, stage) {
    player.server.runCommandSilent(`/astages ${subcommand} "${player.username}" ${stage} true true`)
  }

  function hasStage(player, stage) {
    return !!(player.stages && player.stages.has(stage))
  }

  function hasAnyStage(player, stageList) {
    return stageList.some(stage => hasStage(player, stage))
  }

  function addStageIfMissing(player, stage) {
    if (hasStage(player, stage)) return
    runAStagesCommand(player, 'add', stage)
  }

  function removeStageIfPresent(player, stage) {
    if (!hasStage(player, stage)) return
    runAStagesCommand(player, 'remove', stage)
  }

  function ensureBaseStage(player) {
    const hasIronEquivalent = hasAnyStage(player, IRON_EQUIVALENT_STAGES)
    const hasHigherEquivalent = hasAnyStage(player, HIGHER_EQUIVALENT_STAGES)

    // Keep ore-access progression consistent:
    // any post-primitive age should satisfy age1 ore gates.
    if (hasIronEquivalent || hasHigherEquivalent) {
      addStageIfMissing(player, IRON_STAGE)
    }

    // Primitive stage should only exist before iron era.
    if (hasIronEquivalent || hasHigherEquivalent) {
      removeStageIfPresent(player, BASE_STAGE)
    } else {
      addStageIfMissing(player, BASE_STAGE)
    }

    // Clean legacy aliases only when present to avoid chat spam from AStages.
    for (const legacyStage of LEGACY_STAGES) {
      removeStageIfPresent(player, legacyStage)
    }
  }

  function scheduleBaseStageChecks(event, player) {
    ensureBaseStage(player)

    event.server.scheduleInTicks(10, () => ensureBaseStage(player))
    event.server.scheduleInTicks(40, () => ensureBaseStage(player))
    event.server.scheduleInTicks(120, () => ensureBaseStage(player))
  }

  PlayerEvents.loggedIn(event => {
    scheduleBaseStageChecks(event, event.player)
  })

  PlayerEvents.respawned(event => {
    scheduleBaseStageChecks(event, event.player)
  })

  PlayerEvents.tick(event => {
    const player = event.player
    if (player.tickCount % 200 !== 0) return
    ensureBaseStage(player)
  })
})()
